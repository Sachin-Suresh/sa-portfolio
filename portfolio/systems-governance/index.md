<!-- INTERACTIVE NAVIGATION -->

<script>

document.addEventListener(
  "DOMContentLoaded",
  function () {


    const panels =
      Array.from(
        document.querySelectorAll(
          ".case-panel"
        )
      );


    const navigatorButtons =
      Array.from(
        document.querySelectorAll(
          ".navigator-link"
        )
      );


    const overviewButtons =
      Array.from(
        document.querySelectorAll(
          "[data-open-panel]"
        )
      );



    /*
     * Switch active content panel
     */

    function openPanel(panelName) {


      panels.forEach(
        function (panel) {

          panel.classList.toggle(
            "active",
            panel.dataset.panel ===
              panelName
          );

        }
      );


      navigatorButtons.forEach(
        function (button) {

          button.classList.toggle(
            "active",
            button.dataset.targetPanel ===
              panelName
          );

        }
      );


      /*
       * Keep URL useful without jumping page
       */

      if (history.replaceState) {

        history.replaceState(
          null,
          "",
          "#" + panelName
        );

      }

    }



    /*
     * Right navigator
     */

    navigatorButtons.forEach(
      function (button) {

        button.addEventListener(
          "click",
          function () {

            openPanel(
              button.dataset.targetPanel
            );

          }
        );

      }
    );



    /*
     * Overview cards
     */

    overviewButtons.forEach(
      function (button) {

        button.addEventListener(
          "click",
          function () {

            openPanel(
              button.dataset.openPanel
            );

          }
        );

      }
    );



    /*
     * Open panel from URL hash.
     *
     * Example:
     * /systems-governance/#workflows
     */

    const initialPanel =
      window.location.hash
        .replace("#", "");


    const validPanel =
      panels.some(
        function (panel) {

          return panel.dataset.panel ===
            initialPanel;

        }
      );


    if (validPanel) {

      openPanel(
        initialPanel
      );

    }

    else {

      openPanel(
        "overview"
      );

    }

  }
);

</script>
